\subsection*{Reinforcement Learning and ML in Execution}

\begin{model} \hlt{Stochastic-Control Execution View}\\
Stochastic-control execution chooses a trading rate $v_t$ from state variables such as price $S_t$, remaining inventory $Q_t$, cash $X_t$, and forecast market flow $\mu_t$. A compact sell-side value function is
\begin{align*}
H_t(v)=\mathbb{E}_t\left[X_T^v+Q_T^v(S_T-AQ_T^v)-\phi\int_t^T(Q_s^v)^2ds\right]
\end{align*}
Here $H_t(v)$ is the time-$t$ value of control $v$, $\mathbb{E}_t$ is conditional expectation at time $t$, $T$ is the terminal time, and the superscript $v$ indicates the state path generated by that control. $A$ penalises terminal inventory and $\phi$ penalises carrying inventory through time.
\end{model}

\begin{model} \hlt{Cartea-Jaimungal State Dynamics}\\
One useful specification lets exogenous market flow $\mu_t$ interact with the trader's own rate $v_t$:
\begin{align*}
dS_t&=b_t(\mu_t-v_t)dt+\sigma dW_t, & \widehat{S}_t&=S_t-\left(\frac{\psi}{2}+\kappa v_t\right),\\
dQ_t^v&=-v_tdt, & dX_t^v&=\widehat{S}_t v_tdt.
\end{align*}
Here $b_t$ is permanent-impact sensitivity, $\sigma$ is price-diffusion volatility, $W_t$ is a Brownian shock, $\widehat{S}_t$ is the execution price, $\psi$ is the bid-ask spread, and $\kappa$ is temporary-impact slope.
\end{model}

\begin{principle} \hlt{Value Function Intuition}\\
The control trades off three terms: realised cash, leftover inventory value, and inventory-risk penalty. A larger $\phi$ or $A$ pushes faster liquidation; stronger temporary impact $\kappa$ pushes smoother trading.
\end{principle}

\begin{taxonomy} \hlt{Control vs Learning}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{stochastic control}: specify dynamics and solve for an optimal feedback policy;
\item \hlt{reinforcement learning}: learn the policy or value function from simulated or historical interaction;
\item \hlt{hybrid execution}: use control structure for constraints and ML for forecasts, fills, and regime detection.
\end{enumerate}
\end{taxonomy}
