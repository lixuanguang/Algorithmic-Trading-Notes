\subsubsection{Limit Order Book Mechanics}

\begin{definition} \hlt{Best Quotes and Spread}\\
The \hlt{best bid} is the highest displayed buy price and the \hlt{best ask} is the lowest displayed sell price. The \hlt{bid-ask spread} is $a_t-b_t$, and the quoted midpoint is $m_t=(a_t+b_t)/2$.
\end{definition}

\begin{principle} \hlt{Liquidity Supply and Demand}\\
Resting limit orders supply immediacy to other traders; marketable orders demand it.\\
A standing quote is therefore both a price and a service: it lets someone else trade now.
\end{principle}

\begin{taxonomy} \hlt{Limit Price Placement Segmentation}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{marketable}: crosses the opposite best quote and can execute immediately;
\item \hlt{at the market}: joins the current best bid or ask;
\item \hlt{inside the market}: improves the best bid or ask and narrows the spread;
\item \hlt{behind the market}: rests away from the best quote;
\item \hlt{far from the market}: carries low execution probability but low immediacy cost.
\end{enumerate}
Aggressiveness raises fill probability and lowers waiting time, but it gives up more price concession.
\end{taxonomy}

\begin{proposition} \hlt{Standing Limit Orders Are Trading Options}\\
Let $q$ be order size, $L$ the limit price, $\tau$ the cancellation or expiry time, and $V_t$ the contra-side trader's value at time $t$. While live, a sell limit order gives the market a call-like payoff $q(V_t-L)^+$; a buy limit order gives the market a put-like payoff $q(L-V_t)^+$, where $x^+=\max\{x,0\}$.
\begin{proof}
For $t\leq\tau$, an incoming buyer exercises a sell limit only when buying at $L$ gives non-negative surplus, so the exercise value is $\max\{q(V_t-L),0\}$. The buy-limit case is symmetric, with exercise value $\max\{q(L-V_t),0\}$.
\end{proof}
\end{proposition}

\begin{caveat} \hlt{Limit-Order Regret}\\
Limit orders reduce price concession but introduce two risks. The order may not execute and the trader must chase the price later; or it may execute exactly because better-informed flow is trading through it.
\end{caveat}

\begin{principle} \hlt{Equilibrium Spread as Order-Choice Indifference} \cite{harris_2002}\\
In a continuous order-driven market, the spread adjusts until the marginal trader is indifferent between demanding liquidity with a marketable order and supplying liquidity with a limit order.\\
If spreads are too wide, more traders post limits and compete them down; if spreads are too narrow, passive traders withdraw and marketable flow widens them.
\end{principle}

\begin{mechanism} \hlt{Limit Orders Give Timing Options}\\
A standing limit order gives other traders the option to trade later at a stale price. The option is more valuable when volatility is high, cancellation is slow, informed traders are active, or the contra-side trader can monitor the book faster than the resting order owner.
\end{mechanism}

\begin{definition} \hlt{Tick Size}\\
Tick size is the minimum price increment for limit orders.\\
It discretises prices and can be uniform, price-banded, or liquidity-banded.
\end{definition}

\begin{empiricalfinding} \hlt{Price Clustering}\\
Even when the grid permits many prices, quotes often cluster at round numbers. Clustering reflects communication, search, and queueing convenience, especially when the tick is small relative to price.
\end{empiricalfinding}

\begin{mechanism} \hlt{Binding Tick Constraint}\\
When the tick is large relative to the latent spread, the quoted spread is floored at one tick. Once the tick no longer binds, further reductions mostly change queue behaviour.
\end{mechanism}

\begin{model} \hlt{Harris Tick Discretisation Model}\\
Let $Ask_{j,t}$ and $Bid_{j,t}$ be the best ask and bid for stock $j$ on trade $t$ of day $d$, and let $\Psi_{j,t}=(Ask_{j,t}-Bid_{j,t})/tick_{j,d}$ be the observed spread in ticks. The model assumes an unobserved continuous spread $\widetilde{\Psi}_{j,t}$, then rounds it through a discretisation map $\Psi_{j,t}=\Phi(\widetilde{\Psi}_{j,t})$.\\
The tick is therefore not just a unit of account; it censors small spreads into one-tick quotes.
\end{model}

\begin{definition} \hlt{Continuous Spread Expectation}\\
With turnover proxy $T_j$, volatility $\sigma_{j,d}$, price $P_{j,d}$, and tick size $tick_{j,d}$,\\
a compact specification is $\mathbb{E}[\widetilde{\Psi}_{j,t}\mid T_j,\sigma_{j,d}] =g(T_j,\sigma_{j,d})\frac{P_{j,d}}{tick_{j,d}}$, $g(T_j,\sigma_{j,d})=\alpha T_j^\beta\sigma_{j,d}^{\gamma}$.\\
Here $g$ is the latent relative spread, and $\alpha$, $\beta$, and $\gamma$ are fitted spread-response parameters.\\
Multiplying by $P/tick$ converts it into tick units.
\end{definition}

\begin{mechanism} \hlt{Tick-Size Counterfactual}\\
To estimate the effect of a smaller tick, keep the latent spread distribution fixed and change only the rounding grid. If $\delta$ is the tick-reduction factor and the new tick is $tick/\delta$, an observed $k$-tick spread becomes roughly $k/\delta$ old-tick units before re-rounding.\\
The largest expected spread decrease occurs when current spreads are near one tick.
\end{mechanism}

\begin{taxonomy} \hlt{Harris Model Inputs}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{liquidity}: turnover $T_j$, usually lowering latent spread;
\item \hlt{risk}: volatility $\sigma_{j,d}$, usually raising latent spread;
\item \hlt{price grid}: $P_{j,d}/tick_{j,d}$, controlling how strongly ticks discretise spreads;
\item \hlt{distribution}: a chosen positive density for latent continuous spreads;
\item \hlt{fit check}: compare observed relative spreads with fitted discrete-spread expectations.
\end{enumerate}
\end{taxonomy}

\begin{principle} \hlt{Tick Size and Queue Discipline}\\
The tick defines how expensive it is to step ahead of an existing limit order.\\
Large ticks protect time priority by making queue jumping costly; small ticks make one-tick improvement cheap and weaken the value of being first in queue.
\end{principle}

\begin{mechanism} \hlt{Quoted Depth Under Small Ticks}\\
When the price grid becomes finer, liquidity providers can spread orders across more price levels. Best-limit depth often falls, order sizes shrink, and displayed liquidity must be refreshed more frequently.
\end{mechanism}

\begin{caveat} \hlt{Order-to-Trade Ratio}\\
A smaller tick can increase cancellations and modifications relative to trades. More messages do not necessarily mean more economic liquidity; they may simply reflect shorter quote life and more frequent repricing.
\end{caveat}

\begin{mechanism} \hlt{Cumulative Depth}\\
To compare book quality across tick regimes, measure the cost of executing a fixed monetary size by rebuilding cumulative depth around the best bid and offer. Top-of-book spread alone ignores hidden liquidity, resiliency, and depth beyond the first level.
\end{mechanism}

\begin{principle} \hlt{CLOB as Price-Formation Device}\\
A central limit order book stores unmatched buying and selling interest until compatible prices meet. Each transaction therefore both transfers inventory and updates the public state of price formation.
\end{principle}

\begin{taxonomy} \hlt{Order-Book Visibility}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{open book}: broad participants can observe standing interest;
\item \hlt{closed book}: only the broker, dealer, or venue sees resting orders;
\item \hlt{top-of-book}: only the best limits are visible;
\item \hlt{depth book}: multiple levels, aggregate size, or order-level detail are visible.
\end{enumerate}
\end{taxonomy}

\begin{mechanism} \hlt{LOB Event Types}\\
Order-book dynamics are driven by three event families: insertions add liquidity, cancellations remove liquidity without trading, and marketable orders remove liquidity through transactions. Price changes occur when the best queue is depleted or a better price is inserted.
\end{mechanism}

\begin{taxonomy} \hlt{Tick-Size Quality Proxies}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{binding probability}: share of time or trades with spread equal to one tick;
\item \hlt{spread leeway}: spread in ticks minus one;
\item \hlt{depth}: displayed and cumulative size near the midpoint;
\item \hlt{message intensity}: order-to-trade ratio, cancellations, and modifications.
\end{enumerate}
\end{taxonomy}

\begin{caveat} \hlt{Value of a Quote}\\
As ticks and displayed sizes shrink, quotes can become shorter-lived and less informative. A consolidated tape of trades cannot fully replace quote data if only the best limit is visible and deeper book information is lost.
\end{caveat}

\begin{caveat} \hlt{RFQ Can Mimic a Book}\\
In RFQ markets, dealers answer bilateral requests rather than post firm multilateral quotes.\\
Aggregating several responses can look like a book, but the liquidity is conditional on the request, relationship, size, and information embedded in the enquiry.
\end{caveat}
