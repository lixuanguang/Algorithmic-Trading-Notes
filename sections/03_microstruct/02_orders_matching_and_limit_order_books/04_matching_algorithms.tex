\subsubsection{Matching Algorithms}

\begin{principle} \hlt{Matching Is Market Design}\\
Matching rules decide which orders trade, at what price, and in what priority. Small rule differences can change displayed liquidity, queue value, auction participation, and the usefulness of a venue for large orders.
\end{principle}

\begin{process} \hlt{Rule-Based Matching Procedure}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item rank buy and sell orders by the venue's precedence rules;
\item pair the highest-ranked buy with the highest-ranked sell when their prices cross;
\item price the trade using the venue's trade-pricing rule;
\item leave any residual quantity in the book unless the order instruction cancels it.
\end{enumerate}
The matching loop stops when the best bid is below the best offer, or when no eligible order remains.
\end{process}

\begin{mechanism} \hlt{Continuous Matching}\\
In continuous trading, an incoming marketable order executes immediately against resting interest. The trade-off is speed versus noise: prices update continuously, but liquidity can be thin when order flow is asynchronous.
\end{mechanism}

\begin{mechanism} \hlt{Call Matching}\\
In a call, orders are batched until an uncrossing time and matched at one auction price. The batch creates a thicker crossing point, making calls useful at openings, closings, expiries, and volatility interruptions.
\end{mechanism}

\begin{mechanism} \hlt{Frequent Batch Auctions}\\
Frequent batch auctions group arrivals into short calls. Price priority remains, but within-batch time priority is muted, reducing pure latency value.
\end{mechanism}

\begin{taxonomy} \hlt{Trade-Pricing Rule Segmentation}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{uniform}: all trades from the same call execute at one clearing price;
\item \hlt{discriminatory}: each trade prices from the standing order, accepted quote, or local match;
\item \hlt{derivative}: trades use an external reference price, such as midpoint, primary close, or VWAP.
\end{enumerate}
The same matched quantity can therefore distribute surplus differently across traders.
\end{taxonomy}

\begin{principle} \hlt{Pricing-Rule Incidence}\\
Discriminatory pricing favours aggressive size; uniform pricing favours resting orders by giving all auction fills the same clearing price.
\end{principle}

\begin{process} \hlt{Call Auction Price Rules}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{maximum executable volume}: choose prices that match the largest quantity;
\item \hlt{minimum surplus}: among those prices, minimise unmatched quantity;
\item \hlt{market pressure}: if surplus is only on the buy side choose the higher candidate, and if only on the sell side choose the lower candidate;
\item \hlt{reference price}: if still tied, choose the candidate closest to the venue reference price.
\end{enumerate}
If buy and sell interest do not cross, no auction price can be formed.
\end{process}

\begin{definition} \hlt{Auction Surplus}\\
Auction surplus is the quantity left unmatched at a candidate clearing price. It is not excess volume in the day; it is the residual buy or sell pressure at the auction price.
\end{definition}

\begin{model} \hlt{Single-Price Supply and Demand Schedules}\\
For candidate price $p$, let $S(p)=\sum\limits_i q_i^s\mathbf{1}_{\{s_i\leq p\}}$ and $D(p)=\sum\limits_j q_j^b\mathbf{1}_{\{b_j\geq p\}}$.\\
Matched volume is $M(p)=\min\{S(p),D(p)\}$ and excess demand is $E(p)=D(p)-S(p)$.\\
A single-price auction chooses $p^\star\in\arg\max\limits_{p} M(p)$, then applies imbalance, pressure, or reference-price tie-breaks.
\end{model}

\begin{lemma} \hlt{Trade Price Redistributes Surplus}\\
If a buyer with value $v_b$ trades with a seller with value $v_s$ at any price $p\in[v_s,v_b]$, total trader surplus is $v_b-v_s$ and does not depend on $p$.
\begin{proof}
Buyer surplus is $v_b-p$ and seller surplus is $p-v_s$. Their sum is $v_b-v_s$.
\end{proof}
\end{lemma}

\begin{proposition} \hlt{Price Priority Maximizes Auction Surplus}\\
Unit-expand orders. Sort buyers $b_1\geq b_2\geq\cdots$ and sellers $s_1\leq s_2\leq\cdots$. The surplus-maximizing auction trades the largest $Q$ with $b_Q\geq s_Q$, matching the $Q$ highest buyers with the $Q$ lowest sellers.
\begin{proof}
Exchange any included lower buyer for an excluded higher buyer, and vice versa. Each exchange weakly raises surplus. Additional units are beneficial exactly while $b_Q\geq s_Q$.
\end{proof}
\end{proposition}

\begin{marketrule} \hlt{Randomised Uncrossing}\\
Some venues randomise the exact end of the call after the theoretical auction time. This weakens last-second order placement and cancellation, but it also makes the final book state less deterministic for participants.
\end{marketrule}

\begin{taxonomy} \hlt{Priority Segmentation}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{price priority}: better-priced orders execute first;
\item \hlt{FIFO time priority}: earlier orders at the same price execute first;
\item \hlt{pro-rata priority}: fills at one price are allocated partly by displayed size;
\item \hlt{size priority}: larger orders receive explicit rank or allocation advantages;
\item \hlt{venue-specific priority}: hidden, displayed, broker, or auction-only flags may change rank.
\end{enumerate}
\end{taxonomy}

\begin{marketrule} \hlt{Public Order Precedence}\\
At equal prices, some venues rank public customer orders ahead of member or dealer proprietary interest. This limits access-based queue jumping.
\end{marketrule}

\begin{model} \hlt{Pro-Rata Allocation}\\
At one price level, suppose same-priority resting orders have displayed sizes $q_1,\ldots,q_n$ and compete for incoming quantity $Q<\sum\limits_j q_j$. Under pure pro-rata allocation, order $i$ receives $f_i=Qq_i/\sum\limits_j q_j$ before venue-specific rounding, minimum-fill, and residual rules.
\end{model}

\begin{definition} \hlt{Theoretical Matching Curve}\\
During a call, the theoretical matching curve tracks how close the current indicative matched volume is to the final auction matched volume. It reveals whether the book builds smoothly, jumps at information times, or overshoots before orders are cancelled.
\end{definition}

\begin{caveat} \hlt{Indicative Volume Is Strategic}\\
Pre-auction indicative prices, volumes, and imbalances are signals, not commitments. A book can look deeper than the final auction if traders test information by submitting orders they later cancel or modify.
\end{caveat}
