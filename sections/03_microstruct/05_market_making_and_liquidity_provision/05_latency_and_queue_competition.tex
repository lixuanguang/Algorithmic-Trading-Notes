\subsubsection{Latency and Queue Competition}

\begin{principle} \hlt{Latency as Queue Option}\\
Speed buys queue priority. It is most valuable when queues are thin, stale, or information-sensitive.
\end{principle}

\begin{model} \hlt{HFMM and Investor Race Model}\\
Split participants into high-frequency market makers and longer-horizon investors.\\
A market maker earns spread surplus $s\Delta$ net of maker fee $c_M$; an investor earns value surplus $\Gamma$ net of spread and taker fee $c_T$: $\pi_M=s\Delta-c_M$, $\pi_T=\Gamma-s\Delta-c_T$, $\bar{c}=c_M+c_T$.\\
Here $s$ is spread in ticks, $\Delta$ is tick size, $\Gamma$ is investor gross value gain, $c_M,c_T$ are maker/taker fees, $\pi_M,\pi_T$ are net per-trade gains, and $\bar{c}$ is venue revenue per trade.
\end{model}

\begin{mechanism} \hlt{Race Intensity}\\
Let $\mu_m$ be market maker $m$'s arrival intensity and $\tau_n$ investor $n$'s arrival intensity, with averages $\bar{\mu}$ and $\bar{\tau}$.\\
If technology costs are quadratic, $\Pi_M(m)=\frac{\mu_m}{\bar{\mu}}\pi_M\mathcal{R}-\frac{1}{2}\beta\mu_m^2$, and $\Pi_T(n)=\frac{\tau_n}{\bar{\tau}}\pi_T\mathcal{R}-\frac{1}{2}\gamma\tau_n^2$.\\
Here $\mathcal{R}$ is trade rate, $\Pi_M,\Pi_T$ are total expected gains, and $\beta,\gamma$ are speed-cost coefficients.\\
Lower cost raises optimal speed.
\end{mechanism}

\begin{taxonomy} \hlt{Latency-Cost Segmentation}
\begin{enumerate}[label=\roman*.]
\setlength{\itemsep}{0pt}
\item \hlt{market-maker cost}: co-location, feed handling, quoting engines, and hedge automation;
\item \hlt{investor cost}: broker connectivity, routing latency, algo responsiveness, and monitoring;
\item \hlt{venue fee split}: maker fee or rebate versus taker fee;
\item \hlt{surplus split}: spread capture for makers versus value capture for investors;
\item \hlt{competition level}: number of fast makers relative to investor flow.
\end{enumerate}
\end{taxonomy}

\begin{definition} \hlt{Efficient Maker Fee}\\
In the stylised fee-choice model, the efficient maker fee satisfies $c_M^\star=s\Delta-\frac{\Gamma-\bar{c}}{1+\left(\frac{M}{N}\frac{\gamma}{\beta}\right)^{1/3}}$, $\bar{c}=c_T^\star+c_M^\star$.\\
Here $M,N$ are the numbers of market makers and investors.\\
Negative $c_M^\star$ is a maker rebate; negative $c_T^\star$ is a taker rebate.
\end{definition}

\begin{caveat} \hlt{Fee Rebates Are Comparative}\\
Maker rebates fit scarce or costly market-making; taker rebates fit crowded makers and costly investor access.
\end{caveat}

\begin{practitionernote} \hlt{Queue Competition Diagnostic}\\
Do not classify a fee schedule as good or bad in isolation. Segment by spread size, maker count, investor urgency, queue stability, latency cost, and whether rebates improve displayed liquidity or merely pay for fleeting quotes.
\end{practitionernote}
